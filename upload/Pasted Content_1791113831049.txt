        ndc.set(((e.clientX - r.left) / r.width) * 2 - 1, -((e.clientY - r.top) / r.height) * 2 + 1);
        ray.setFromCamera(ndc, camera);
        const hit = ray.intersectObject(hitPlane)[0];
        if (!hit) return;
        const p = pin(GREEN, 0.5);
        p.position.copy(world.worldToLocal(hit.point.clone()));
        p.rotation.y = Math.random() * 6.28;
        world.add(p);
        popIn(p, 0, 0.5);
        dropped.push(p);
        if (dropped.length > 8) world.remove(dropped.shift());
        pinN++;
        const chip = document.getElementById("pinChip");
        chip.textContent = pinN + " pin";
        chip.classList.remove("hidden");
      }
      canvas.addEventListener("pointerup", (e) => { if (!movedFar) dropPin(e); });

      /* --- siang / malam --- */
      let night = false;
      const daySky = new THREE.Color(0xfdfcf7), nightSky = new THREE.Color(0x35544a);
      const dayGnd = new THREE.Color(0xcfc9ba), nightGnd = new THREE.Color(0x14231e);
      const daySun = new THREE.Color(0xffffff), nightSun = new THREE.Color(0xa9c7e8);

      const dnBtn = document.getElementById("dayNight");
      dnBtn.addEventListener("click", () => {
        night = !night;
        dnBtn.querySelector(".ic-moon").classList.toggle("hidden", night);
        dnBtn.querySelector(".ic-sun").classList.toggle("hidden", !night);
        document.getElementById("dnLabel").textContent = night ? "Siang" : "Malam";
        wrap.closest(".scene-card").classList.toggle("night", night);
      });

      function size() {
        const w = wrap.clientWidth, h = wrap.clientHeight;
        renderer.setSize(w, h, false);
        const aspect = w / h;
        camera.aspect = aspect; camera.updateProjectionMatrix();
        camZ = aspect < 0.8 ? 17 : aspect < 1.3 ? 15 : 13.4;
      }
      new ResizeObserver(size).observe(wrap); size();

      let t = 0, mix = 0, last = performance.now();
      function frame(now) {
        const dt = Math.min((now - last) / 1000, 0.05); last = now; t += dt;

        if (!dragging) {
          targetRotY += vel; vel *= 0.93;
          idle += dt;
          if (idle > 2.5) targetRotY += dt * 0.12;
        }
        rotY += (targetRotY - rotY) * 0.12;
        world.rotation.y = rotY;

        parallax.x += (parallax.tx - parallax.x) * 0.05;
        parallax.y += (parallax.ty - parallax.y) * 0.05;
        camera.position.set(camBase.x + parallax.x * 1.4, camBase.y - parallax.y * 0.8, camZ);
        camera.lookAt(0, 0.8, 0);

        mix += ((night ? 1 : 0) - mix) * 0.04;
        hemi.color.lerpColors(daySky, nightSky, mix);
        hemi.groundColor.lerpColors(dayGnd, nightGnd, mix);
        hemi.intensity = L(1.05, 0.5, mix);
        sun.color.lerpColors(daySun, nightSun, mix);
        sun.intensity = L(1.35, 0.55, mix);
        glowMat.emissiveIntensity = mix * 1.9;
        sunBall.position.y = L(5.7, 1.1, mix);
        sunMat.emissiveIntensity = L(0.95, 0.05, mix);
        moonBall.position.y = L(1.1, 5.5, mix);
        moonMat.emissiveIntensity = L(0.06, 0.85, mix);
        starMat.opacity = mix * 0.9;

        if (!reduced) {
          mainPin.position.y = 0.08 + Math.sin(t * 1.8) * 0.12;
          mainPin.rotation.y = t * 0.6;
          const a = t * 0.35;
          carG.position.set(Math.cos(a) * R, 0.05, Math.sin(a) * R);
          carG.rotation.y = -a - Math.PI / 2;
          clouds.forEach(c => {
            c.g.position.x = c.bx + Math.sin(t * c.spd + c.ph) * 0.8;
            c.g.position.y = c.by + Math.sin(t * 0.6 + c.ph) * 0.12;
          });
          foliage.forEach((f, i) => { f.rotation.z = Math.sin(t * 1.3 + i * 1.7) * 0.06; });
        }
        popUpdate(dt);
        renderer.render(scene, camera);
        requestAnimationFrame(frame);
      }
      requestAnimationFrame(frame);

    } catch (err) {
      document.getElementById("sceneWrap")?.closest(".scene-card")?.classList.add("hidden");
      console.warn("3D tidak dapat dimuat:", err);
    }
