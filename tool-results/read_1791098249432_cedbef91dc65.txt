   620→   620→function CountUp({ to, decimals = 0 }: { to: number; decimals?: number }) {
   621→   621→  const ref = useRef<HTMLSpanElement>(null);
   622→   622→  const inView = useInView(ref, { once: true, margin: "-60px" });
   623→   623→  const reduce = useReducedMotion();
   624→   624→  const [val, setVal] = useState(0);
   625→   625→
   626→   626→  useEffect(() => {
   627→   627→    if (!inView) return;
   628→   628→    if (reduce) { setVal(to); return; }
   629→   629→    const dur = 1400, t0 = performance.now();
   630→   630→    let raf = 0;
   631→   631→    const tick = (now: number) => {
   632→   632→      const x = Math.min((now - t0) / dur, 1);
   633→   633→      setVal(to * (1 - Math.pow(1 - x, 3)));
   634→   634→      if (x < 1) raf = requestAnimationFrame(tick);
   635→   635→    };
   636→   636→    raf = requestAnimationFrame(tick);
   637→   637→    return () => cancelAnimationFrame(raf);
   638→   638→  }, [inView, to, reduce]);
   639→   639→
   640→   640→  return <span ref={ref}>{val.toFixed(decimals)}</span>;
   641→   641→}
